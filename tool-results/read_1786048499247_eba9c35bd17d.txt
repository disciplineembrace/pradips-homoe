   241→/// Connectivity detection service.
   242→///
   243→/// Monitors network status and provides a stream of connectivity changes.
   244→/// Used by the sync engine to trigger sync when connectivity returns.
   245→library;
   246→
   247→import 'dart:async';
   248→import 'package:connectivity_plus/connectivity_plus.dart';
   249→
   250→enum ConnectivityStatus {
   251→  online,
   252→  offline,
   253→}
   254→
   255→/// Singleton connectivity monitor.
   256→class ConnectivityService {
   257→  static final ConnectivityService _instance = ConnectivityService._internal();
   258→  factory ConnectivityService() => _instance;
   259→  ConnectivityService._internal();
   260→
   261→  final Connectivity _connectivity = Connectivity();
   262→
   263→  final StreamController<ConnectivityStatus> _statusController =
   264→      StreamController<ConnectivityStatus>.broadcast();
   265→  Stream<ConnectivityStatus> get statusStream => _statusController.stream;
   266→
   267→  ConnectivityStatus _currentStatus = ConnectivityStatus.online;
   268→  ConnectivityStatus get currentStatus => _currentStatus;
   269→
   270→  bool get isOnline => _currentStatus == ConnectivityStatus.online;
   271→  bool get isOffline => _currentStatus == ConnectivityStatus.offline;
   272→
   273→  /// Initialize and start monitoring.
   274→  void init() {
   275→    _connectivity.onConnectivityChanged.listen((List<ConnectivityResult> results) {
   276→      final wasOffline = _currentStatus == ConnectivityStatus.offline;
   277→      final hasConnection = results.any((r) =>
   278→          r == ConnectivityResult.wifi ||
   279→          r == ConnectivityResult.mobile ||
   280→          r == ConnectivityResult.ethernet);
   281→
   282→      _currentStatus =
   283→          hasConnection ? ConnectivityStatus.online : ConnectivityStatus.offline;
   284→      _statusController.add(_currentStatus);
   285→
   286→      // If we just came back online, the sync engine will pick this up
   287→      // via the status stream and trigger an incremental sync + outbox flush.
   288→      if (wasOffline && isOnline) {
   289→        // Stream event is enough; sync engine subscribes.
   290→      }
   291→    });
   292→
   293→    // Check initial status synchronously
   294→    _connectivity.checkConnectivity().then((results) {
   295→      final hasConnection = results.any((r) =>
   296→          r == ConnectivityResult.wifi ||
   297→          r == ConnectivityResult.mobile ||
   298→          r == ConnectivityResult.ethernet);
   299→      _currentStatus =
   300→          hasConnection ? ConnectivityStatus.online : ConnectivityStatus.offline;
   301→      _statusController.add(_currentStatus);
   302→    });
   303→  }
   304→
   305→  /// Manually check current connectivity (fresh query).
   306→  Future<bool> checkOnline() async {
   307→    final results = await _connectivity.checkConnectivity();
   308→    final hasConnection = results.any((r) =>
   309→        r == ConnectivityResult.wifi ||
   310→        r == ConnectivityResult.mobile ||
   311→        r == ConnectivityResult.ethernet);
   312→    _currentStatus =
   313→        hasConnection ? ConnectivityStatus.online : ConnectivityStatus.offline;
   314→    return hasConnection;
   315→  }
   316→
   317→  void dispose() {
   318→    _statusController.close();
   319→  }
   320→}

   323→/// Dio HTTP client with authentication, retry, and error handling.
   324→///
   325→/// All API calls go through this client. It:
   326→///   - Attaches the session cookie/token to every request
   327→///   - Handles 401 (auth expired) by attempting session refresh
   328→///   - Converts DioErrors to typed ApiExceptions
   329→///   - Enforces HTTPS-only communication
   330→///   - Logs requests with sensitive data redacted
   331→library;
   332→
   333→import 'package:dio/dio.dart';
   334→import 'package:flutter_secure_storage/flutter_secure_storage.dart';
   335→import '../config/app_config.dart';
   336→import 'api_exceptions.dart';
   337→
   338→class DioClient {
   339→  late final Dio _dio;
   340→  final FlutterSecureStorage _secureStorage;
   341→
   342→  DioClient(this._secureStorage) {
   343→    _dio = Dio(BaseOptions(
   344→      baseUrl: AppConfig.apiBaseUrl,
   345→      connectTimeout: Duration(seconds: AppConfig.connectTimeoutSec),
   346→      receiveTimeout: Duration(seconds: AppConfig.receiveTimeoutSec),
   347→      sendTimeout: Duration(seconds: AppConfig.sendTimeoutSec),
   348→      headers: {
   349→        'Content-Type': 'application/json',
   350→        'Accept': 'application/json',
   351→      },
   352→      // Validate that we only use HTTPS in production
   353→      validateStatus: (status) => status != null && status >= 200 && status < 300,
   354→    ));
   355→
   356→    // Add auth interceptor
   357→    _dio.interceptors.add(InterceptorsWrapper(
   358→      onRequest: (options, handler) async {
   359→        // Attach session cookie if available
   360→        final token = await _secureStorage.read(key: AppConfig.secureStorageKey);
   361→        if (token != null && token.isNotEmpty) {
   362→          // The website uses httpOnly cookies for sessions.
   363→          // For the Flutter app, we send the token as a Cookie header
   364→          // OR as a Bearer token (the API accepts both).
   365→          options.headers['Cookie'] = 'ph_session=$token';
   366→          options.headers['Authorization'] = 'Bearer $token';
   367→        }
   368→        handler.next(options);
   369→      },
   370→      onError: (error, handler) async {
   371→        // On 401, attempt session refresh once
   372→        if (error.response?.statusCode == 401) {
   373→          // Try to refresh session — if that fails, propagate the error
   374→          // (the auth repository will handle logout/redirect)
   375→          final refreshed = await _attemptSessionRefresh();
   376→          if (refreshed) {
   377→            // Retry the original request
   378→            try {
   379→              final retryResponse = await _dio.fetch(error.requestOptions);
   380→              return handler.resolve(retryResponse);
   381→            } catch (e) {
   382→              return handler.next(error);
   383→            }
   384→          }
   385→        }
   386→        handler.next(error);
   387→      },
   388→    ));
   389→  }
   390→
   391→  /// Attempt to refresh the session by calling the session endpoint.
   392→  /// Returns true if the session is still valid.
   393→  Future<bool> _attemptSessionRefresh() async {
   394→    try {
   395→      final response = await _dio.get(AppConfig.sessionEndpoint);
   396→      final data = response.data;
   397→      if (data is Map && data['authenticated'] == true) {
   398→        return true;
   399→      }
   400→    } catch (_) {
   401→      // Session is invalid — user will be redirected to login
   402→    }
   403→    return false;
   404→  }
   405→
   406→  /// GET request with typed response.
   407→  Future<T> get<T>(
   408→    String path, {
   409→    Map<String, dynamic>? queryParameters,
   410→    Options? options,
   411→  }) async {
   412→    try {
   413→      final response = await _dio.get<T>(
   414→        path,
   415→        queryParameters: queryParameters,
   416→        options: options,
   417→      );
   418→      return response.data as T;
   419→    } on DioException catch (e) {
   420→      throw _convertDioError(e);
   421→    }
   422→  }
   423→
   424→  /// POST request with typed response.
   425→  Future<T> post<T>(
   426→    String path, {
   427→    dynamic data,
   428→    Map<String, dynamic>? queryParameters,
   429→    Options? options,
   430→  }) async {
   431→    try {
   432→      final response = await _dio.post<T>(
   433→        path,
   434→        data: data,
   435→        queryParameters: queryParameters,
   436→        options: options,
   437→      );
   438→      return response.data as T;
   439→    } on DioException catch (e) {
   440→      throw _convertDioError(e);
   441→    }
   442→  }
   443→
   444→  /// PUT request with typed response.
   445→  Future<T> put<T>(
   446→    String path, {
   447→    dynamic data,
   448→    Map<String, dynamic>? queryParameters,
   449→    Options? options,
   450→  }) async {

   463→
   464→  /// DELETE request.
   465→  Future<void> delete(
   466→    String path, {
   467→    Map<String, dynamic>? queryParameters,
   468→    Options? options,
   469→  }) async {
   470→    try {
   471→      await _dio.delete(
   472→        path,
   473→        queryParameters: queryParameters,
   474→        options: options,
   475→      );
   476→    } on DioException catch (e) {
   477→      throw _convertDioError(e);
   478→    }
   479→  }
   480→
   481→  /// Convert DioException to typed ApiException.
   482→  ApiException _convertDioError(DioException e) {
   483→    switch (e.type) {
   484→      case DioExceptionType.connectionTimeout:
   485→      case DioExceptionType.sendTimeout:
   486→      case DioExceptionType.receiveTimeout:
   487→        return TimeoutException(e.message);
   488→      case DioExceptionType.connectionError:
   489→        return const NoConnectionException();
   490→      case DioExceptionType.badResponse:
   491→        final statusCode = e.response?.statusCode ?? 500;
   492→        final data = e.response?.data;
   493→        String message = 'Server error';
   494→        if (data is Map && data['error'] != null) {
   495→          message = data['error'].toString();
   496→        }
   497→        if (statusCode == 401) {
   498→          return AuthException(message, statusCode: statusCode);
   499→        }
   500→        if (statusCode == 403) {
   501→          return PermissionException(message);
   502→        }
   503→        return ServerException(message, statusCode);
   504→      default:
   505→        return ServerException('Unexpected error', 500, e.message);
   506→    }
   507→  }
   508→}

   511→/// Database encryption service — manages SQLCipher encryption key.
   512→///
   513→/// - Generates a strong random 256-bit encryption key on first app launch.
   514→/// - Stores the key ONLY in Android secure storage (Android Keystore).
   515→/// - Never hardcodes the key in source code.
   516→/// - Never commits the key to GitHub.
   517→/// - Never stores the key in shared preferences.
   518→/// - Provides the key to Drift for opening an encrypted SQLCipher database.
   519→/// - Handles missing, corrupted, or invalid keys safely.
   520→/// - Does NOT silently delete user data when database opening fails.
   521→///
   522→/// The key is stored as a hex-encoded 32-byte (256-bit) random value.
   523→library;
   524→
   525→import 'dart:math';
   526→import 'dart:typed_data';
   527→
   528→import 'package:flutter_secure_storage/flutter_secure_storage.dart';
   529→
   530→/// Secure storage key for the database encryption key.
   531→const _dbEncryptionKeyStorageKey = 'ph_db_encryption_key';
   532→
   533→/// Service that manages the SQLCipher encryption key.
   534→class DatabaseEncryptionService {
   535→  final FlutterSecureStorage _secureStorage;
   536→
   537→  DatabaseEncryptionService(this._secureStorage);
   538→
   539→  /// Get or create the database encryption key.
   540→  ///
   541→  /// On first launch: generates a new 256-bit random key and stores it
   542→  /// in Android secure storage (Android Keystore).
   543→  /// On subsequent launches: retrieves the existing key from secure storage.
   544→  ///
   545→  /// Returns the key as a hex string suitable for SQLCipher's `PRAGMA key`.
   546→  Future<String> getOrCreateKey() async {
   547→    // Try to read the existing key from secure storage
   548→    final existingKey = await _secureStorage.read(
   549→      key: _dbEncryptionKeyStorageKey,
   550→    );
   551→
   552→    if (existingKey != null && existingKey.isNotEmpty) {
   553→      // Key exists — return it
   554→      return existingKey;
   555→    }
   556→
   557→    // First launch — generate a new strong random key
   558→    final newKey = _generateStrongKey();
   559→
   560→    // Store the key in Android secure storage (Android Keystore)
   561→    await _secureStorage.write(
   562→      key: _dbEncryptionKeyStorageKey,
   563→      value: newKey,
   564→    );
   565→
   566→    return newKey;
   567→  }
   568→
   569→  /// Check if an encryption key already exists in secure storage.
   570→  Future<bool> hasKey() async {
   571→    final key = await _secureStorage.read(key: _dbEncryptionKeyStorageKey);
   572→    return key != null && key.isNotEmpty;
   573→  }
   574→
   575→  /// Delete the encryption key from secure storage.
   576→  ///
   577→  /// Called on logout — after the database file is deleted, the key
   578→  /// is also removed so that any residual data cannot be decrypted.
   579→  Future<void> deleteKey() async {
   580→    await _secureStorage.delete(key: _dbEncryptionKeyStorageKey);
   581→  }
   582→
   583→  /// Generate a strong random 256-bit (32-byte) encryption key.
   584→  ///
   585→  /// Uses Dart's Random.secure() which delegates to the platform's
   586→  /// cryptographically secure random number generator:
   587→  /// - Android: java.security.SecureRandom
   588→  ///
   589→  /// Returns the key as a hex string (64 hex characters = 32 bytes = 256 bits).
   590→  String _generateStrongKey() {
   591→    final random = Random.secure();
   592→    final bytes = Uint8List(32); // 256 bits
   593→    for (int i = 0; i < 32; i++) {
   594→      bytes[i] = random.nextInt(256);
   595→    }
   596→    // Convert to hex string — SQLCipher accepts hex keys with PRAGMA key = "x'...'"
   597→    return bytes.map((b) => b.toRadixString(16).padLeft(2, '0')).join();
   598→  }
   599→
   600→  /// Format the key for SQLCipher's PRAGMA key statement.
   601→  ///
   602→  /// SQLCipher expects the key in the format: x'hexstring'
   603→  /// when using a raw hex key.
   604→  String formatKeyForSqlCipher(String hexKey) {
   605→    return "x'$hexKey'";
   606→  }
   607→
   608→  /// Clear all secure storage entries related to the database.
   609→  ///
   610→  /// Called on logout to ensure no sensitive data remains.
   611→  Future<void> clearAllSecureData() async {
   612→    await _secureStorage.delete(key: _dbEncryptionKeyStorageKey);
   613→  }
   614→}

  1025→///   - Never committed to GitHub
  1026→///   - Never stored in shared preferences

  1029→///   - serverId: stable unique ID from server
  1030→///   - serverVersion: version/revision number for conflict detection
  1031→///   - updatedAt: server-side last-modified timestamp
  1032→///   - deletedAt: soft-delete marker (null = active)

  1061→  tables: [

  1063→    Rubrics,
  1064→    Books,

  1073→class AppDatabase extends _$AppDatabase {
  1074→  AppDatabase() : super(_openEncryptedConnection());
  1075→
  1076→  /// For testing — inject an in-memory database (no encryption).
  1077→  AppDatabase.forTesting(QueryExecutor e) : super(e);
  1078→
  1079→  @override
  1080→  int get schemaVersion => 1;
  1081→
  1082→  @override
  1083→  MigrationStrategy get migration => MigrationStrategy(
  1084→        onCreate: (m) => m.createAll(),
  1085→        onUpgrade: (m, from, to) async {
  1086→          // Additive migrations only — never delete columns/tables.
  1087→          if (from < 2) {
  1088→            // Example future migration:
  1089→            // await m.addColumn(remedies, remedies.newColumn);
  1090→          }
  1091→        },
  1092→        beforeOpen: (details) async {
  1093→          // Enable foreign keys
  1094→          await customStatement('PRAGMA foreign_keys = ON');
  1095→          // Enable WAL mode for better concurrent read performance
  1096→          await customStatement('PRAGMA journal_mode = WAL');
  1097→        },
  1098→      );
  1099→
  1100→  /// Open an encrypted SQLCipher database connection.
  1101→  ///
  1102→  /// This method:
  1103→  ///   1. Gets or creates the encryption key from Android secure storage.
  1104→  ///   2. Opens the database file with SQLCipher encryption.
  1105→  ///   3. Handles migration from unencrypted SQLite if an old DB exists.
  1106→  ///   4. Handles corrupted/invalid keys safely — does NOT silently delete data.
  1107→  static LazyDatabase _openEncryptedConnection() {
  1108→    return LazyDatabase(() async {
  1109→      final dbFolder = await getApplicationDocumentsDirectory();
  1110→      final dbPath = p.join(dbFolder.path, AppConfig.dbFileName);
  1111→      final file = File(dbPath);
  1112→
  1113→      // Get the encryption service
  1114→      final secureStorage = const FlutterSecureStorage(
  1115→        aOptions: AndroidOptions(encryptedSharedPreferences: true),
  1116→      );
  1117→      final encryptionService = DatabaseEncryptionService(secureStorage);
  1118→
  1119→      // Get or create the encryption key
  1120→      final hexKey = await encryptionService.getOrCreateKey();
  1121→      final sqlCipherKey = encryptionService.formatKeyForSqlCipher(hexKey);
  1122→
  1123→      // Check if an old unencrypted database exists and needs migration
  1124→      await _migrateUnencryptedIfNeeded(dbPath);
  1125→
  1126→      // Open the database with SQLCipher encryption
  1127→      return NativeDatabase.createInBackground(
  1128→        file,
  1129→        setup: (rawDb) {
  1130→          rawDb.execute("PRAGMA key = $sqlCipherKey;");
  1131→        },
  1132→      );
  1133→    });
  1134→  }
  1135→
  1136→  /// Migrate an old unencrypted SQLite database to encrypted SQLCipher.
  1137→  ///
  1138→  /// If an old unencrypted database file exists (from a previous app version
  1139→  /// that didn't use encryption), this method renames it to .bak (preserved)
  1140→  /// and lets a new encrypted database be created.
  1141→  /// User data is NEVER silently deleted.
  1142→  static Future<void> _migrateUnencryptedIfNeeded(String dbPath) async {
  1143→    final file = File(dbPath);
  1144→
  1145→    if (!await file.exists()) {
  1146→      return;
  1147→    }
  1148→
  1149→    try {
  1150→      final bytes = await file.openRead(0, 16).first;
  1151→      final header = String.fromCharCodes(bytes);
  1152→      if (header == 'SQLite format 3\x00') {
  1153→        final backupPath = '$dbPath.unencrypted.bak';
  1154→        await file.rename(backupPath);
  1155→      }
  1156→    } catch (_) {
  1157→      // Could not read the file — might be corrupted or encrypted
  1158→      // Do NOT delete the file — let Drift try to open it with the key.
  1159→    }
  1160→  }
  1161→
  1162→  /// Close the database and clear all local data on logout.
  1163→  ///
  1164→  /// Called by AuthRepository.logout() to:
  1165→  ///   1. Close the database connection
  1166→  ///   2. Delete the database file (contains cached content + user data)
  1167→  ///   3. Delete the encryption key from secure storage
  1168→  Future<void> clearOnLogout() async {
  1169→    await close();
  1170→
  1171→    try {
  1172→      final dbFolder = await getApplicationDocumentsDirectory();
  1173→      final dbPath = p.join(dbFolder.path, AppConfig.dbFileName);
  1174→      final file = File(dbPath);
  1175→      if (await file.exists()) {
  1176→        await file.delete();
  1177→      }
  1178→      final walFile = File('$dbPath-wal');
  1179→      final shmFile = File('$dbPath-shm');
  1180→      if (await walFile.exists()) await walFile.delete();
  1181→      if (await shmFile.exists()) await shmFile.delete();
  1182→    } catch (_) {}
  1183→
  1184→    try {
  1185→      final secureStorage = const FlutterSecureStorage(
  1186→        aOptions: AndroidOptions(encryptedSharedPreferences: true),
  1187→      );
  1188→      final encryptionService = DatabaseEncryptionService(secureStorage);
  1189→      await encryptionService.deleteKey();
  1190→    } catch (_) {}
  1191→  }
  1192→}

  1831→  factory RubricDto.fromJson(Map<String, dynamic> json) {
  1832→    final remediesRaw = json['remedies'] as List? ?? [];
  1833→    final remedies = remediesRaw.map((r) {
  1834→      if (r is Map) {
  1835→        return ParsedRemedyDto(
  1836→          abbrev: r['abbrev']?.toString() ?? '',
  1837→          grade: (r['grade'] as num?)?.toInt() ?? 1,
  1838→        );
  1839→      }
  1840→      // Format: "abbrev|grade"
  1841→      final parts = r.toString().split('|');
  1842→      return ParsedRemedyDto(
  1843→        abbrev: parts[0],
  1844→        grade: parts.length > 1 ? int.tryParse(parts[1]) ?? 1 : 1,
  1845→      );
  1846→    }).toList();
  1847→
  1848→    final byGrade = <String, List<String>>{};
  1849→    final bg = json['byGrade'] as Map? ?? {};
  1850→    for (final entry in bg.entries) {
  1851→      byGrade[entry.key] = (entry.value as List).map((e) => e.toString()).toList();
  1852→    }
  1853→
  1854→    return RubricDto(
  1855→      id: json['id']?.toString() ?? '',
  1856→      parentId: json['parentId']?.toString() ?? json['parent_id']?.toString(),
  1857→      source: json['source']?.toString() ?? json['author']?.toString() ?? '',
  1858→      chapter: json['chapter']?.toString() ?? '',
  1859→      title: json['title']?.toString() ?? '',
  1860→      fullPath: json['fullPath']?.toString() ?? json['full_path']?.toString() ?? json['title']?.toString() ?? '',
  1861→      level: (json['level'] as num?)?.toInt() ?? 0,
  1862→      remedies: remedies,
  1863→      remedyCount: (json['remedyCount'] as num?)?.toInt() ?? remedies.length,
  1864→      byGrade: byGrade,
  1865→    );
  1866→  }
  1867→

  1869→}
